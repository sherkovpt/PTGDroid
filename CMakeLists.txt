cmake_minimum_required(VERSION 3.20)
project(ptgdroid C)

set(CMAKE_C_STANDARD 11)

# Recompiled mode: C translated from *your own copy* of the game by recomp/armrecomp.py (see README).
# Without it (the default for the published APK) the game's ARM code runs in the interpreter.
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/generated/app/dispatch_table.c)
  set(PTG_RECOMPILED_DEFAULT ON)
else()
  set(PTG_RECOMPILED_DEFAULT OFF)
endif()
option(PTG_RECOMPILED "Link code recompiled from the user's game files (generated/)" ${PTG_RECOMPILED_DEFAULT})

file(GLOB RT_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/runtime/*.c)
list(FILTER RT_SOURCES EXCLUDE REGEX "(main_.*|touch)\\.c$")
set(GEN_SOURCES "")
if(PTG_RECOMPILED)
  file(GLOB GEN_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/generated/app/*.c ${CMAKE_CURRENT_SOURCE_DIR}/generated/zlib/*.c)
  list(FILTER GEN_SOURCES EXCLUDE REGEX "imports_table\\.c$")
endif()
message(STATUS "PTGDroid: recompiled game code ${PTG_RECOMPILED}")

add_library(ptg_core STATIC ${GEN_SOURCES} ${RT_SOURCES})
target_include_directories(ptg_core PUBLIC runtime)
if(PTG_RECOMPILED)
  target_compile_definitions(ptg_core PUBLIC PTG_RECOMPILED)
endif()

if(MSVC)
  target_compile_options(ptg_core PRIVATE /W3 /wd4102 /wd4146 /wd4702 /wd4244 /bigobj /MP)
else()
  target_compile_options(ptg_core PRIVATE -Wno-unused-label -fwrapv)
endif()

set(SDL_DIR ${CMAKE_CURRENT_SOURCE_DIR}/third_party/SDL)
if(ANDROID)
  set(SDL_SHARED ON CACHE BOOL "" FORCE)
  set(SDL_STATIC OFF CACHE BOOL "" FORCE)
else()
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
endif()
set(SDL_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(${SDL_DIR} SDL2 EXCLUDE_FROM_ALL)

if(ANDROID)
  # SDLActivity loads libmain.so and calls SDL_main.
  add_library(main SHARED runtime/main_sdl.c runtime/touch.c)
  target_link_libraries(main ptg_core SDL2::SDL2 log android)
else()
  add_executable(ptg runtime/main_sdl.c runtime/touch.c)
  target_link_libraries(ptg ptg_core SDL2::SDL2-static)
  if(WIN32)
    target_link_libraries(ptg winmm)
  endif()
endif()

if(NOT ANDROID)
  add_executable(ptg_host runtime/main_host.c)
  target_link_libraries(ptg_host ptg_core)
  add_executable(ptg_difftest runtime/main_difftest.c)
  target_link_libraries(ptg_difftest ptg_core)
endif()
